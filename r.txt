best_aic = Inf
best_p = 0
best_q= 0

for (p in 0.3){
  for (q in 0.3){
    model <- try(
      arima(log_ap, order = c(p, 1, q)),
      silent = TRUE
    )
    
    if (!inherits(model, "try-error")) {
      aic <- AIC(model)
      
      if (aic < best_aic) {
        best_aic <- aic
        best_p <- p
        best_q <- q
      }
    }
  }
}

#set4
y <- log(AirPassengers)
orders <- list(c(1,1,1), c(2,1,1), c(0,1,2))
results <- data.frame()

for(o in orders) {
  fit <- arima(y, order=o)
  n <- length(residuals(fit))
  k <- length(fit$coef)
  fpe <- fit$sigma2 * (n+k)/(n-k)
  
  results <- rbind(results, data.frame(
    Model=paste(o, collapse=","),
    LogLik=as.numeric(logLik(fit)),
    AIC=AIC(fit), BIC=BIC(fit), FPE=fpe))
}

#set 6
library(forecast)
library(lmtest)

y <- log(AirPassengers)

# 1. ARIMA(2,1,2): coefficient significance
fit <- arima(y, order=c(2,1,2))
se <- sqrt(diag(fit$var.coef))
z <- fit$coef / se
data.frame(Estimate=fit$coef, SE=se, Z=z,
           P=2*pnorm(-abs(z)))

# 2. Remove non-significant terms; compare AIC
fits <- lapply(list(c(2,1,2), c(0,1,2), c(0,1,1)),
               function(o) arima(y, order=o))
data.frame(Model=c("212","012","011"),
           AIC=sapply(fits, AIC))

# 3. AIC and BIC grid search
r <- expand.grid(p=0:3, q=0:3)
r$AIC <- r$BIC <- NA
for(i in 1:nrow(r)) {
  f <- arima(y, order=c(r$p[i],1,r$q[i]))
  r$AIC[i] <- AIC(f)
  r$BIC[i] <- BIC(f)
}
r[order(r$BIC), ]

# 4. Residual diagnostics for the lowest-BIC model
best <- r[which.min(r$BIC), ]
fit <- arima(y, order=c(best$p,1,best$q))
e <- na.omit(residuals(fit))
dwtest(e ~ 1)
spec.pgram(e, cumulative=TRUE,
           main="Cumulative Periodogram of Residuals")
