# Estimated mathematical equation
phi <- fit_arma$coef["ar1"]
theta <- fit_arma$coef["ma1"]
intercept <- fit_arma$coef["intercept"]

cat("AR coefficient:", phi, "\n")
cat("MA coefficient:", theta, "\n")
cat("Intercept:", intercept, "\n")

cat("Equation: z[t] =", intercept,
    "+", phi, "* z[t-1] + e[t] +",
    theta, "* e[t-1]\n")

# Check AR and MA characteristic roots
ar_roots <- polyroot(c(1, -phi))
ma_roots <- polyroot(c(1, theta))

print(ar_roots)
print(ma_roots)

print(Mod(ar_roots) > 1)
print(Mod(ma_roots) > 1)

best_aic <- Inf
best_p <- 0
best_q <- 0

for (p in 0:3) {
  for (q in 0:3) {
    model <- try(
      arima(log_ap, order=c(p,1,q)),
      silent=TRUE
    )
    
    if (!inherits(model, "try-error")) {
      aic <- AIC(model)
      
      if (aic < best_aic) {
        best_aic <- aic
        best_p <- p
        best_q <- q
      }
    }
  }
}

  fit <- arima(y, order=o)

    Model=paste(o, collapse=","),
    LogLik=as.numeric(logLik(fit)),
    AIC=AIC(fit), BIC=BIC(fit), FPE=fpe))

# Topic 6: ARIMA Parameter Estimation and Order Selection
# Topic 7: FPE, AIC, BIC and Residual Analysis

library(forecast)
library(lmtest)

data("AirPassengers")
y <- log(AirPassengers)

# 1. Fit ARIMA(2,1,2) and test coefficient significance

fit <- arima(y, order=c(2,1,2))

se <- sqrt(diag(fit$var.coef))
z <- fit$coef / se
p <- 2 * pnorm(-abs(z))

significance <- data.frame(
  Estimate=fit$coef,
  SE=se,
  Z=z,
  P=p,
  Significant=p < 0.05
)

print(significance)

# 2. Eliminate non-significant terms and compare models using AIC/BIC

orders <- list(
  c(2,1,2),
  c(1,1,2),
  c(2,1,1),
  c(0,1,2),
  c(1,1,1),
  c(0,1,1)
)

fits <- lapply(orders, function(o) {
  arima(y, order=o)
})

comparison <- data.frame(
  Model=sapply(orders, function(o) paste(o, collapse=",")),
  AIC=sapply(fits, AIC),
  BIC=sapply(fits, BIC)
)

print(comparison)

# 3. Grid search over AR(p) and MA(q) orders
# p = 0 to 3, d = 1, q = 0 to 3

r <- expand.grid(p=0:3, q=0:3)
r$AIC <- NA
r$BIC <- NA

for(i in 1:nrow(r)) {
  f <- try(
    arima(y, order=c(r$p[i],1,r$q[i])),
    silent=TRUE
  )
  
  if(!inherits(f, "try-error")) {
    r$AIC[i] <- AIC(f)
    r$BIC[i] <- BIC(f)
  }
}

r$Model <- paste0("ARIMA(", r$p, ",1,", r$q, ")")
r$Order <- r$p + r$q

# Display models sorted by BIC
r <- r[order(r$BIC), ]
print(r)

# Best models according to AIC and BIC
print("Best model according to AIC:")
print(r[which.min(r$AIC), ])

print("Best model according to BIC:")
print(r[which.min(r$BIC), ])

# 4. Residual diagnostics for the lowest-BIC model

best <- r[which.min(r$BIC), ]

fit <- arima(y, order=c(best$p,1,best$q))
print(fit)

e <- na.omit(residuals(fit))

# Durbin-Watson test
dw_result <- dwtest(e ~ 1)
print(dw_result)

# Cumulative periodogram
spec.pgram(
  e,
  cumulative=TRUE,
  main="Cumulative Periodogram of Residuals"
)

# Additional residual diagnostic plots
tsdisplay(e, main="Residual Diagnostics")
